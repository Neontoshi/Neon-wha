pub fn update_badge(app: &tauri::AppHandle, count: u32) {
    if let Some(tray) = app.tray_by_id("main-tray") {
        let icon_path = if count > 0 {
            "icons/tray/unread.png"
        } else {
            "icons/tray/idle.png"
        };
        if let Ok(icon) = tauri::image::Image::from_path(icon_path) {
            let _ = tray.set_icon(Some(icon));
        }
        let tooltip = if count > 0 {
            format!("NeonWhat — {} unread", count)
        } else {
            "NeonWhat".to_string()
        };
        let _ = tray.set_tooltip(Some(&tooltip));
    }
}

use serde_json::{json, Value};
use std::fs;
use std::path::PathBuf;
use tauri::Manager;

fn config_path(app: &tauri::AppHandle) -> PathBuf {
    app.path()
        .app_config_dir()
        .unwrap_or_else(|_| PathBuf::from("."))
        .join("neonwhat.json")
}

pub fn load(app: &tauri::AppHandle) -> Value {
    let path = config_path(app);
    if let Ok(data) = fs::read_to_string(&path) {
        serde_json::from_str(&data).unwrap_or_else(|_| default_config())
    } else {
        default_config()
    }
}

pub fn save(app: &tauri::AppHandle, config: &Value) -> Result<(), Box<dyn std::error::Error>> {
    let path = config_path(app);
    if let Some(parent) = path.parent() {
        fs::create_dir_all(parent)?;
    }
    fs::write(&path, serde_json::to_string_pretty(config)?)?;
    Ok(())
}

// UPDATED: Safely load the custom theme CSS from the app's config directory
pub fn load_custom_css(app: &tauri::AppHandle) -> String {
    let mut path = app
        .path()
        .app_config_dir()
        .unwrap_or_else(|_| PathBuf::from("."));
    path.push("assets");
    path.push("themes");
    path.push("example.css");

    // If the file doesn't exist, return an empty string instead of panicking
    fs::read_to_string(path).unwrap_or_default()
}

fn default_config() -> Value {
    json!({
        "theme": "default",
        "close_to_tray": true,
        "notifications_enabled": true,
        "shortcut_toggle": "Ctrl+Shift+W"
    })
}

pub fn run() {
    tauri::Builder::default()
        .plugin(tauri_plugin_single_instance::init(|app, _args, _cwd| {
            if let Some(win) = app.get_webview_window("main") {
                let _ = win.show();
                let _ = win.set_focus();
            }
        }))
        .plugin(tauri_plugin_notification::init())
        .plugin(tauri_plugin_global_shortcut::Builder::new().build())
        .invoke_handler(tauri::generate_handler![
            commands::notify_show,
            commands::unread_set,
            commands::open_settings,
            commands::get_config,
            commands::save_config,
        ])
        .setup(|app| {
            window::build_main_window(app)?;
            tray::setup_tray(app)?;
            shortcuts::setup_shortcuts(app)?;
            Ok(())
        })
        .on_window_event(|window, event| {
            if let tauri::WindowEvent::CloseRequested { api, .. } = event {
                if window.label() == "main" {
                    api.prevent_close();
                    let _ = window.hide();
                }
            }
        })
        .run(tauri::generate_context!())
        .expect("error while running NeonWhat");
}

    app.notification()
        .permission_state()
        .map(|state| state == tauri_plugin_notification::PermissionState::Granted)
        .unwrap_or(false)

/// Get the data directory for the webview.
/// If a profile name is provided, it creates a subfolder for it (e.g., "profile-work").
pub fn get_session_dir(app: &tauri::AppHandle, profile: Option<&str>) -> Option<PathBuf> {
    let mut path = app.path().app_data_dir().ok()?;

    if let Some(p) = profile {
        path.push(format!("profile-{}", p));
    }

    // Create the directory if it doesn't exist
    if !path.exists() {
        let _ = std::fs::create_dir_all(&path);
    }

    Some(path)
}

use tauri::Manager;
use tauri_plugin_global_shortcut::{Code, GlobalShortcutExt, Modifiers, Shortcut, ShortcutState};

pub fn setup_shortcuts(app: &tauri::App) -> Result<(), Box<dyn std::error::Error>> {
    let toggle = Shortcut::new(Some(Modifiers::CONTROL | Modifiers::SHIFT), Code::KeyW);
    let app_handle = app.handle().clone();

    app.global_shortcut()
        .on_shortcut(toggle, move |_app, _shortcut, event| {
            if event.state() == ShortcutState::Pressed {
                if let Some(w) = app_handle.get_webview_window("main") {
                    if w.is_visible().unwrap_or(false) {
                        let _ = w.hide();
                    } else {
                        let _ = w.show();
                        let _ = w.set_focus();
                    }
                }
            }
        })?;

    Ok(())
}

use tauri::{
    menu::{Menu, MenuItem, PredefinedMenuItem},
    tray::{MouseButton, MouseButtonState, TrayIconBuilder, TrayIconEvent},
    Manager,
};

pub fn setup_tray(app: &tauri::App) -> Result<(), Box<dyn std::error::Error>> {
    let show = MenuItem::with_id(app, "show", "Show Window", true, None::<&str>)?;
    let settings = MenuItem::with_id(app, "settings", "Settings", true, None::<&str>)?;
    let sep = PredefinedMenuItem::separator(app)?;
    let quit = MenuItem::with_id(app, "quit", "Quit", true, None::<&str>)?;
    let menu = Menu::with_items(app, &[&show, &settings, &sep, &quit])?;

    let _tray = TrayIconBuilder::with_id("main-tray")
        .icon(
            app.default_window_icon()
                .cloned()
                .unwrap_or_else(|| tauri::image::Image::from_path("icons/icon.png").unwrap()),
        )
        .menu(&menu)
        .show_menu_on_left_click(false)
        .on_menu_event(|app, event| match event.id.as_ref() {
            "show" => {
                if let Some(w) = app.get_webview_window("main") {
                    let _ = w.show();
                    let _ = w.set_focus();
                }
            }
            "settings" => {
                crate::commands::open_settings(app.clone());
            }
            "quit" => app.exit(0),
            _ => {}
        })
        .on_tray_icon_event(|tray, event| {
            if let TrayIconEvent::Click {
                button: MouseButton::Left,
                button_state: MouseButtonState::Up,
                ..
            } = event
            {
                let app = tray.app_handle();
                if let Some(w) = app.get_webview_window("main") {
                    if w.is_visible().unwrap_or(false) {
                        let _ = w.hide();
                    } else {
                        let _ = w.show();
                        let _ = w.set_focus();
                    }
                }
            }
        })
        .build(app)?;

    Ok(())
}

use tauri::{WebviewUrl, WebviewWindowBuilder};

const CHROME_UA: &str = "Mozilla/5.0 (X11; Linux x86_64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/130.0.0.0 Safari/537.36";

pub fn build_main_window(app: &tauri::App) -> Result<(), Box<dyn std::error::Error>> {
    // 1. Load the custom CSS and wrap it in a script to inject into the page
    let custom_css = crate::config::load_custom_css(app.handle());
    let css_inject_script = format!(
        r#"
        (function() {{
            function applyCss() {{
                const style = document.createElement('style');
                style.id = 'neonwhat-custom-css';
                style.textContent = `{}`;
                document.head.appendChild(style);
            }}
            if (document.readyState === 'loading') {{
                window.addEventListener('DOMContentLoaded', applyCss);
            }} else {{
                applyCss();
            }}
        }})();
        "#,
        custom_css.replace('`', "\\`")
    );

    // 2. Build the window (Removed the custom data_directory here)
    WebviewWindowBuilder::new(
        app,
        "main",
        WebviewUrl::External("https://web.whatsapp.com".parse()?),
    )
    .title("NeonWhat")
    .user_agent(CHROME_UA)
    .initialization_script(include_str!("../inject/notification-bridge.js"))
    .initialization_script(include_str!("../inject/title-watcher.js"))
    .initialization_script(&css_inject_script)
    .inner_size(1200.0, 800.0)
    .build()?;

    Ok(())
}
