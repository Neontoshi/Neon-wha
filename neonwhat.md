### `inject/inject.js`

```javascript
// NeonWhat injected script: notification bridge + unread badge watcher.
(function () {
  if (window.top !== window) return; // main frame only
  if (window.__neonwhat_injected) return;
  window.__neonwhat_injected = true;

  var warned = false;

  // Returns true if the call was dispatched, false if IPC isn't available.
  function call(cmd, args) {
    var invoke = window.__TAURI__ && window.__TAURI__.core && window.__TAURI__.core.invoke;
    if (!invoke) {
      if (!warned) {
        warned = true;
        console.warn("[NeonWhat] Tauri IPC unavailable: check withGlobalTauri and capabilities");
      }
      return false;
    }
    invoke(cmd, args).catch(function (e) {
      console.warn("[NeonWhat] " + cmd + " failed", e);
    });
    return true;
  }

  // ---- Notification bridge ----
  function NeonWhatNotification(title, options) {
    options = options || {};
    var self = new EventTarget(); // so addEventListener/dispatchEvent/onclick don't throw
    self.title = String(title);
    self.body = String(options.body || "");
    self.tag = options.tag || "";
    self.onclick = null;
    self.close = function () {};
    call("notify_show", { title: self.title, body: self.body });
    return self;
  }
  NeonWhatNotification.permission = "granted";
  NeonWhatNotification.requestPermission = function () {
    return Promise.resolve("granted");
  };
  try {
    window.Notification = NeonWhatNotification;
  } catch (e) {
    // Notification may be non-writable in WebKitGTK
  }

  // ---- Unread count from document.title, e.g. "(3) WhatsApp" ----
  var last = -1;

  function parseUnread(title) {
    var m = /^\((\d+)\)/.exec(title || "");
    return m ? parseInt(m[1], 10) : 0;
  }

  function report() {
    var count = parseUnread(document.title);
    if (count === last) return;
    if (call("unread_set", { count: count })) last = count;
  }

  function observeTitle() {
    new MutationObserver(report).observe(document.head, {
      childList: true,
      subtree: true,
      characterData: true,
    });
    report();
  }

  if (document.readyState === "loading") {
    window.addEventListener("DOMContentLoaded", observeTitle);
  } else {
    observeTitle();
  }
})();
```

```rust
pub fn update_badge(app: &tauri::AppHandle, count: u32) {
    if let Some(tray) = app.tray_by_id("main-tray") {
        let bytes: &[u8] = if count > 0 {
            include_bytes!("../icons/tray/unread.png")
        } else {
            include_bytes!("../icons/tray/idle.png")
        };
        if let Ok(icon) = tauri::image::Image::from_bytes(bytes) {
            let _ = tray.set_icon(Some(icon));
        }
        let tooltip = if count > 0 {
            format!("NeonWhat — {} unread", count)
        } else {
            "NeonWhat".to_string()
        };
        let _ = tray.set_tooltip(Some(&tooltip));
    }
}
```

```rust
#[tauri::command]
pub fn notify_show(app: tauri::AppHandle, title: String, body: String) {
    let enabled = crate::config::load(&app)["notifications_enabled"]
        .as_bool()
        .unwrap_or(true);
    if enabled && crate::notifications::is_permission_granted(&app) {
        crate::notifications::send_notification(&app, &title, &body);
    }
}

#[tauri::command]
pub fn unread_set(app: tauri::AppHandle, count: u32) {
    if let Some(win) = app.get_webview_window("main") {
        let title = if count > 0 {
            format!("({}) NeonWhat", count)
        } else {
            "NeonWhat".to_string()
        };
        let _ = win.set_title(&title);
    }
    crate::badge::update_badge(&app, count);
}

#[tauri::command]
pub fn open_settings(app: tauri::AppHandle) {
    if let Some(win) = app.get_webview_window("settings") {
        let _ = win.show();

        return;
    }
    let _ = tauri::WebviewWindowBuilder::new(
        &app,
        "settings",
        tauri::WebviewUrl::App("index.html".into()),

    .title("NeonWhat Settings")
    .inner_size(500.0, 400.0)
```

```rust
use serde_json::{json, Value};
use std::fs;
use std::path::PathBuf;
use tauri::Manager;

fn config_path(app: &tauri::AppHandle) -> PathBuf {
    app.path()
        .app_config_dir()
        .unwrap_or_else(|_| PathBuf::from("."))
        .join("neonwhat.json")
}

pub fn load(app: &tauri::AppHandle) -> Value {
    let path = config_path(app);
    if let Ok(data) = fs::read_to_string(&path) {
        serde_json::from_str(&data).unwrap_or_else(|_| default_config())
    } else {
        default_config()
    }
}

pub fn save(app: &tauri::AppHandle, config: &Value) -> Result<(), Box<dyn std::error::Error>> {
    let path = config_path(app);
    if let Some(parent) = path.parent() {
        fs::create_dir_all(parent)?;
    }
    fs::write(&path, serde_json::to_string_pretty(config)?)?;
    Ok(())
}

// UPDATED: Safely load the custom theme CSS from the app's config directory
pub fn load_custom_css(app: &tauri::AppHandle) -> String {

    path.push("assets");
    path.push("themes");

    // If the file doesn't exist, return an empty string instead of panicking

}
```

```rust
use tauri::Manager;

pub fn run() {
    tauri::Builder::default()
        .plugin(tauri_plugin_single_instance::init(|app, _args, _cwd| {
            if let Some(win) = app.get_webview_window("main") {
                let _ = win.show();
                let _ = win.set_focus();
            }
        }))
        .plugin(tauri_plugin_notification::init())
        .plugin(tauri_plugin_global_shortcut::Builder::new().build())
        .invoke_handler(tauri::generate_handler![
            commands::notify_show,
            commands::unread_set,
            commands::open_settings,
            commands::get_config,
            commands::save_config,
        ])
        .setup(|app| {
            window::build_main_window(app)?;
            tray::setup_tray(app)?;
            shortcuts::setup_shortcuts(app)?;
            Ok(())
        })
        .on_window_event(|window, event| {
            if let tauri::WindowEvent::CloseRequested { api, .. } = event {
                if window.label() == "main"
                    && config::load(window.app_handle())["close_to_tray"]
                        .as_bool()
                        .unwrap_or(true)
                {
                    api.prevent_close();
                    let _ = window.hide();
                }
            }
        })
        .run(tauri::generate_context!())
        .expect("error while running NeonWhat");
}
```

```rust
use tauri::AppHandle;
use tauri_plugin_notification::NotificationExt;

fn escape_markup(s: &str) -> String {
    s.replace('&', "&amp;")
        .replace('<', "&lt;")
        .replace('>', "&gt;")
}

/// Send a native OS notification.
pub fn send_notification(app: &AppHandle, title: &str, body: &str) {
    let _ = app
        .notification()
        .builder()
        .title(title)
        .body(escape_markup(body))
        .show();
}

/// Check if the OS currently allows notifications from this app.
pub fn is_permission_granted(app: &AppHandle) -> bool {
    app.notification()
        .permission_state()
        .map(|state| state == tauri_plugin_notification::PermissionState::Granted)
        .unwrap_or(false)
}
```

```rust
use std::path::PathBuf;
use tauri::Manager;

/// Get the data directory for the webview.
/// If a profile name is provided, it creates a subfolder for it (e.g., "profile-work").
pub fn get_session_dir(app: &tauri::AppHandle, profile: Option<&str>) -> Option<PathBuf> {
    let mut path = app.path().app_data_dir().ok()?;

    if let Some(p) = profile {
        path.push(format!("profile-{}", p));
    }

    // Create the directory if it doesn't exist
    if !path.exists() {
        let _ = std::fs::create_dir_all(&path);
    }

    Some(path)
}
```

### `src/shortcuts.rs`

```rust
use tauri::Manager;
use tauri_plugin_global_shortcut::{Code, GlobalShortcutExt, Modifiers, Shortcut, ShortcutState};

pub fn setup_shortcuts(app: &tauri::App) -> Result<(), Box<dyn std::error::Error>> {
    let toggle = Shortcut::new(Some(Modifiers::CONTROL | Modifiers::SHIFT), Code::KeyW);
    let app_handle = app.handle().clone();

    app.global_shortcut()
        .on_shortcut(toggle, move |_app, _shortcut, event| {
            if event.state() == ShortcutState::Pressed {
                if let Some(w) = app_handle.get_webview_window("main") {
                    if w.is_visible().unwrap_or(false) {
                        let _ = w.hide();
                    } else {
                        let _ = w.show();
                        let _ = w.set_focus();
                    }
                }
            }
        })
        .unwrap_or_else(|e| eprintln!("global shortcut not registered: {e}"));
    Ok(())
}
```

```rust
use tauri::{WebviewUrl, WebviewWindowBuilder};

const CHROME_UA: &str = "Mozilla/5.0 (X11; Linux x86_64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/156.0.0.0 Safari/537.36";

pub fn build_main_window(app: &tauri::App) -> Result<(), Box<dyn std::error::Error>> {
    // 1. Load the custom CSS and wrap it in a script to inject into the page
    let custom_css = crate::config::load_custom_css(app.handle());
    let css_literal = serde_json::to_string(&custom_css)?;

    let css_inject_script = format!(
        r#"
        (function() {{
            function applyCss() {{
                const style = document.createElement('style');
                style.id = 'neonwhat-custom-css';
                style.textContent = {};

                document.head.appendChild(style);
            }}
            if (document.readyState === 'loading') {{
                window.addEventListener('DOMContentLoaded', applyCss);
            }} else {{
                applyCss();
            }}
        }})();
        "#,
        css_literal
    );

    // 2. Build the window (Removed the custom data_directory here)
    //
    #[allow(unused_variables)]
    let win = WebviewWindowBuilder::new(
        app,
        "main",
        WebviewUrl::External("https://web.whatsapp.com".parse()?),
    )
    .title("NeonWhat")
    .user_agent(CHROME_UA)
    .initialization_script(include_str!("../inject/inject.js"))
    .initialization_script(&css_inject_script)
    .inner_size(1200.0, 800.0)
    .build()?;

    // Let status/video audio play without a user gesture (WebKitGTK)
    #[cfg(target_os = "linux")]
    win.with_webview(|wv| {
        use webkit2gtk::{SettingsExt, WebViewExt};
        if let Some(s) = wv.inner().settings() {
            s.set_media_playback_requires_user_gesture(false);
        }
    })?;
```
